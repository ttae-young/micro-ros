#define LOWFAT_NODE_COUNT       2U
#define LOWFAT_REGION_SHIFT     12U
#define LOWFAT_REGION_SIZE      (1UL << LOWFAT_REGION_SHIFT)
#define LOWFAT_POOL_SIZE        (LOWFAT_NODE_COUNT * LOWFAT_REGION_SIZE)
#define LOWFAT_ALIGNMENT        32U
#define NODE_PUBLISHER          0U
#define NODE_SUBSCRIBER         1U
#define ATTACK_STACK_WORDS      256U
#define MICROROS_STACK_WORDS    4096U

typedef struct {
    rcl_node_t node;
    rcl_publisher_t publisher;
    std_msgs__msg__Int32 message;
    volatile uint32_t published_count;
    volatile uint32_t received_count;
    uintptr_t subscriber_message_address;
} publisher_region_t;

typedef struct {
    rcl_node_t node;
    rcl_subscription_t subscription;
    std_msgs__msg__Int32 message;
} subscriber_region_t;

extern UART_HandleTypeDef hlpuart1;

bool cubemx_transport_open(struct uxrCustomTransport *transport);
bool cubemx_transport_close(struct uxrCustomTransport *transport);
size_t cubemx_transport_write(struct uxrCustomTransport *transport,
                              const uint8_t *buf, size_t len, uint8_t *err);
size_t cubemx_transport_read(struct uxrCustomTransport *transport,
                             uint8_t *buf, size_t len, int timeout,
                             uint8_t *err);
void *microros_allocate(size_t size, void *state);
void microros_deallocate(void *pointer, void *state);
void *microros_reallocate(void *pointer, size_t size, void *state);
void *microros_zero_allocate(size_t count, size_t size, void *state);

static uint8_t g_lowfat_pool[LOWFAT_POOL_SIZE]
    __attribute__((aligned(LOWFAT_REGION_SIZE)));
static size_t g_lowfat_next[LOWFAT_NODE_COUNT];
static publisher_region_t *g_pub;
static subscriber_region_t *g_sub;
static StackType_t g_attack_stack[ATTACK_STACK_WORDS]
    __attribute__((aligned(LOWFAT_ALIGNMENT)));
static TaskHandle_t g_attack_task_handle;

static rclc_support_t g_support;
static rcl_timer_t g_timer;
static rclc_executor_t g_executor;

static void print_reset_cause(void)
{
    uint32_t csr = RCC->CSR;
    SEGGER_RTT_printf(0, "[DIAG] Reset flags: 0x%08lx\n", (unsigned long)csr);
    if ((csr & RCC_CSR_SFTRSTF) != 0U) {
        SEGGER_RTT_WriteString(0, "[DIAG] Reset cause: SOFTWARE RESET\n");
    } else if ((csr & RCC_CSR_IWDGRSTF) != 0U) {
        SEGGER_RTT_WriteString(0, "[DIAG] Reset cause: INDEPENDENT WATCHDOG\n");
    } else if ((csr & RCC_CSR_WWDGRSTF) != 0U) {
        SEGGER_RTT_WriteString(0, "[DIAG] Reset cause: WINDOW WATCHDOG\n");
    } else if ((csr & RCC_CSR_BORRSTF) != 0U) {
        SEGGER_RTT_WriteString(0, "[DIAG] Reset cause: BROWNOUT/POWER RESET\n");
    } else if ((csr & RCC_CSR_PINRSTF) != 0U) {
        SEGGER_RTT_WriteString(0, "[DIAG] Reset cause: RESET PIN\n");
    } else {
        SEGGER_RTT_WriteString(0, "[DIAG] Reset cause: OTHER\n");
    }
    __HAL_RCC_CLEAR_RESET_FLAGS();
}

static uintptr_t region_start(uint32_t node)
{
    return (uintptr_t)&g_lowfat_pool[node * LOWFAT_REGION_SIZE];
}

static uintptr_t region_end(uint32_t node)
{
    return region_start(node) + LOWFAT_REGION_SIZE - 1U;
}

static void *lowfat_alloc(uint32_t node, size_t size)
{
    size_t aligned;
    size_t offset;
    void *result;

    if ((node >= LOWFAT_NODE_COUNT) || (size == 0U)) {
        return NULL;
    }
    aligned = (size + LOWFAT_ALIGNMENT - 1U) & ~(LOWFAT_ALIGNMENT - 1U);
    offset = g_lowfat_next[node];
    if ((offset + aligned) > LOWFAT_REGION_SIZE) {
        return NULL;
    }
    result = (void *)(region_start(node) + offset);
    g_lowfat_next[node] = offset + aligned;
    memset(result, 0, aligned);
    return result;
}

static void stop_with_error(const char *stage, rcl_ret_t status)
{
    SEGGER_RTT_printf(0, "[ERROR] %s failed (%ld)\n", stage, (long)status);
    for (;;) {
        __NOP();
    }
}

#define CHECK_RCL(call, stage) do {                 \
    rcl_ret_t _status = (call);                     \
    if (_status != RCL_RET_OK) {                    \
        stop_with_error((stage), _status);          \
    }                                               \
} while (0)

static void timer_callback(rcl_timer_t *timer, int64_t last_call_time)
{
    (void)last_call_time;
    if (timer == NULL) {
        return;
    }

    /* Publish 1 first, then 2, 3, ... */
    g_pub->message.data++;
    SEGGER_RTT_WriteString(0, "\n[Publisher]\n");
    SEGGER_RTT_printf(0, "Publisher value : %ld\n",
                     (long)g_pub->message.data);
    SEGGER_RTT_printf(0, "Subscriber value: %ld\n",
                     (long)g_sub->message.data);
    SEGGER_RTT_printf(0, "Publisher object address : %p\n", (void *)g_pub);
    SEGGER_RTT_printf(0, "Subscriber message address: %p\n",
                     (void *)&g_sub->message.data);
    if (rcl_publish(&g_pub->publisher, &g_pub->message, NULL) == RCL_RET_OK) {
        g_pub->published_count++;
        SEGGER_RTT_WriteString(0, "\n[Unauthorized direct access attempt]\n");
        SEGGER_RTT_printf(0, "Publisher object address : %p\n", (void *)g_pub);
        SEGGER_RTT_printf(0, "Target Subscriber address: %p\n",
                         (void *)g_pub->subscriber_message_address);
        xTaskNotifyGive(g_attack_task_handle);
    } else {
        SEGGER_RTT_WriteString(0, "[Publisher] Publish failed\n");
    }
}

static void subscription_callback(const void *message)
{
    const std_msgs__msg__Int32 *received =
        (const std_msgs__msg__Int32 *)message;

    g_pub->received_count++;
    SEGGER_RTT_WriteString(0, "\n[Subscriber received]\n");
    SEGGER_RTT_printf(0, "Publisher value : %ld\n",
                     (long)g_pub->message.data);
    SEGGER_RTT_printf(0, "Subscriber value: %ld\n", (long)received->data);
    SEGGER_RTT_printf(0, "Subscriber message address: %p\n",
                     (const void *)&received->data);
}

/* The faulting LDR is exactly one 16-bit instruction.  The MemManage
 * handler skips it so the next one-second Pub/Sub cycle can continue. */
__attribute__((naked, noinline))
static void forbidden_read_once(const volatile int32_t *address)
{
    (void)address;
    __asm volatile(
        "ldr r0, [r0] \n"
        "bx lr        \n");
}

static void publisher_attack_task(void *argument)
{
    publisher_region_t *pub = (publisher_region_t *)argument;

    for (;;) {
        (void)ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
        forbidden_read_once(
            (volatile const int32_t *)pub->subscriber_message_address);
    }
}

static void micro_ros_task(void *argument)
{
    (void)argument;
    rcl_allocator_t allocator = rcl_get_default_allocator();

    rmw_uros_set_custom_transport(
        true, &hlpuart1,
        cubemx_transport_open,
        cubemx_transport_close,
        cubemx_transport_write,
        cubemx_transport_read);

    allocator.allocate = microros_allocate;
    allocator.deallocate = microros_deallocate;
    allocator.reallocate = microros_reallocate;
    allocator.zero_allocate = microros_zero_allocate;
    if (!rcutils_set_default_allocator(&allocator)) {
        SEGGER_RTT_WriteString(0, "[ERROR] Allocator setup failed\n");
        for (;;) { __NOP(); }
    }

    SEGGER_RTT_WriteString(0, "[micro-ROS] Waiting for agent...\n");
    while (rmw_uros_ping_agent(1000, 1) != RMW_RET_OK) {
        HAL_Delay(1000);
    }
    SEGGER_RTT_WriteString(0, "[micro-ROS] Agent connected\n");

    CHECK_RCL(rclc_support_init(&g_support, 0, NULL, &allocator), "support");

    g_pub->node = rcl_get_zero_initialized_node();
    g_sub->node = rcl_get_zero_initialized_node();
    g_pub->publisher = rcl_get_zero_initialized_publisher();
    g_sub->subscription = rcl_get_zero_initialized_subscription();
    g_timer = rcl_get_zero_initialized_timer();
    g_executor = rclc_executor_get_zero_initialized_executor();

    CHECK_RCL(rclc_node_init_default(
        &g_pub->node, "node_a", "", &g_support), "Node A");
    CHECK_RCL(rclc_node_init_default(
        &g_sub->node, "node_b", "", &g_support), "Node B");
    CHECK_RCL(rclc_publisher_init_default(
        &g_pub->publisher, &g_pub->node,
        ROSIDL_GET_MSG_TYPE_SUPPORT(std_msgs, msg, Int32),
        "shared_int32_topic"), "publisher");
    CHECK_RCL(rclc_subscription_init_default(
        &g_sub->subscription, &g_sub->node,
        ROSIDL_GET_MSG_TYPE_SUPPORT(std_msgs, msg, Int32),
        "shared_int32_topic"), "subscription");
    CHECK_RCL(rclc_timer_init_default(
        &g_timer, &g_support, RCL_MS_TO_NS(1000), timer_callback), "timer");
    CHECK_RCL(rclc_executor_init(
        &g_executor, &g_support.context, 2, &allocator), "executor");
    CHECK_RCL(rclc_executor_add_timer(&g_executor, &g_timer), "add timer");
    CHECK_RCL(rclc_executor_add_subscription(
        &g_executor, &g_sub->subscription, &g_sub->message,
        subscription_callback, ON_NEW_DATA), "add subscription");

    g_pub->message.data = 0;
    SEGGER_RTT_WriteString(0, "[micro-ROS] Pub/Sub started (1 second period)\n");
    for (;;) {
        (void)rclc_executor_spin_some(&g_executor, RCL_MS_TO_NS(100));
    }
}

void MPU_Demo_Start(void)
{
    SEGGER_RTT_Init();
    print_reset_cause();
    memset(g_lowfat_pool, 0, sizeof(g_lowfat_pool));
    memset(g_lowfat_next, 0, sizeof(g_lowfat_next));

    g_pub = (publisher_region_t *)lowfat_alloc(
        NODE_PUBLISHER, sizeof(publisher_region_t));
    g_sub = (subscriber_region_t *)lowfat_alloc(
        NODE_SUBSCRIBER, sizeof(subscriber_region_t));
    if ((g_pub == NULL) || (g_sub == NULL)) {
        SEGGER_RTT_WriteString(0, "[ERROR] LowFat allocation failed\n");
        for (;;) { __NOP(); }
    }

    g_pub->subscriber_message_address = (uintptr_t)&g_sub->message.data;
    SEGGER_RTT_WriteString(0, "\n=== LOWFAT REGIONS ===\n");
    SEGGER_RTT_printf(0, "[Region 0] Publisher : 0x%08lx - 0x%08lx (4 KiB)\n",
        (unsigned long)region_start(NODE_PUBLISHER),
        (unsigned long)region_end(NODE_PUBLISHER));
    SEGGER_RTT_printf(0, "[Region 1] Subscriber: 0x%08lx - 0x%08lx (4 KiB)\n",
        (unsigned long)region_start(NODE_SUBSCRIBER),
        (unsigned long)region_end(NODE_SUBSCRIBER));
    SEGGER_RTT_printf(0, "[Publisher] object=%p\n", (void *)g_pub);
    SEGGER_RTT_printf(0, "[Subscriber] message=%p\n",
                     (void *)&g_sub->message.data);
    SEGGER_RTT_WriteString(0, "[Initial values]\n");
    SEGGER_RTT_printf(0, "Publisher value : %ld\n", (long)g_pub->message.data);
    SEGGER_RTT_printf(0, "Subscriber value: %ld\n", (long)g_sub->message.data);
    const TaskParameters_t attack_parameters = {
        .pvTaskCode = publisher_attack_task,
        .pcName = "PubAttack",
        .usStackDepth = ATTACK_STACK_WORDS,
        .pvParameters = g_pub,
        .uxPriority = tskIDLE_PRIORITY + 4U,
        .puxStackBuffer = g_attack_stack,
        .xRegions = {
            {
                .pvBaseAddress = (void *)region_start(NODE_PUBLISHER),
                .ulLengthInBytes = LOWFAT_REGION_SIZE,
                .ulParameters = tskMPU_REGION_READ_WRITE |
                                tskMPU_REGION_EXECUTE_NEVER
            },
            { 0 }, { 0 }
        }
    };

    if (xTaskCreateRestricted(&attack_parameters,
                              &g_attack_task_handle) != pdPASS) {
        SEGGER_RTT_WriteString(0, "[ERROR] Attack task creation failed\n");
        for (;;) { __NOP(); }
    }
    if (xTaskCreate(micro_ros_task, "microROS", MICROROS_STACK_WORDS,
                    NULL, (tskIDLE_PRIORITY + 3U) | portPRIVILEGE_BIT,
                    NULL) != pdPASS) {
        SEGGER_RTT_WriteString(0, "[ERROR] micro-ROS task creation failed\n");
        for (;;) { __NOP(); }
    }
}

void MemManage_Handler_C(uint32_t *fault_stack)
    __attribute__((used, noinline));

void MemManage_Handler(void) __attribute__((naked));
void MemManage_Handler(void)
{
    __asm volatile(
        "tst lr, #4       \n"
        "ite eq           \n"
        "mrseq r0, msp    \n"
        "mrsne r0, psp    \n"
        "b MemManage_Handler_C\n");
}

void HardFault_Handler_C(uint32_t *fault_stack)
    __attribute__((used, noinline));

void HardFault_Handler(void) __attribute__((naked));
void HardFault_Handler(void)
{
    __asm volatile(
        "tst lr, #4       \n"
        "ite eq           \n"
        "mrseq r0, msp    \n"
        "mrsne r0, psp    \n"
        "b HardFault_Handler_C\n");
}

void HardFault_Handler_C(uint32_t *fault_stack)
{
    SEGGER_RTT_WriteString(0, "\n[DIAG] HARDFAULT\n");
    SEGGER_RTT_printf(0, "Fault PC     : 0x%08lx\n",
                     (unsigned long)fault_stack[6]);
    SEGGER_RTT_printf(0, "CFSR         : 0x%08lx\n",
                     (unsigned long)SCB->CFSR);
    SEGGER_RTT_printf(0, "HFSR         : 0x%08lx\n",
                     (unsigned long)SCB->HFSR);
    for (;;) { __NOP(); }
}

void vAssertCalled(const char *file, int line)
{
    taskDISABLE_INTERRUPTS();
    SEGGER_RTT_WriteString(0, "\n[DIAG] FREERTOS ASSERT\n");
    SEGGER_RTT_printf(0, "File         : %s\n", file);
    SEGGER_RTT_printf(0, "Line         : %d\n", line);
    for (;;) { __NOP(); }
}

void vApplicationStackOverflowHook(TaskHandle_t task, char *task_name)
{
    (void)task;
    SEGGER_RTT_WriteString(0, "\n[DIAG] STACK OVERFLOW\n");
    SEGGER_RTT_printf(0, "Task         : %s\n", task_name);
    for (;;) { __NOP(); }
}

void MemManage_Handler_C(uint32_t *fault_stack)
{
    const uint32_t blocked_address = SCB->MMFAR;

    SEGGER_RTT_WriteString(0, "\n[MPU] Cross-region read blocked\n");
    SEGGER_RTT_printf(0, "Publisher object address : %p\n", (void *)g_pub);
    SEGGER_RTT_printf(0, "Requested memory address : 0x%08lx\n",
                     (unsigned long)g_pub->subscriber_message_address);
    SEGGER_RTT_printf(0, "MPU reported fault address: 0x%08lx\n",
                     (unsigned long)blocked_address);
    SEGGER_RTT_printf(0, "MPU register address      : %p\n", (void *)MPU);
    SEGGER_RTT_WriteString(0, "Result           : ACCESS DENIED\n");
    fault_stack[6] += 2U; /* Skip the 16-bit faulting LDR. */
    SCB->CFSR = (SCB->CFSR & 0xFFU);
    __DSB();
    __ISB();
}
